<script>
    document.addEventListener("DOMContentLoaded", function() {
        const carousel = document.getElementById('reelsCarousel');
        const track = document.getElementById('reelsTrack');
        
        // 1. Create the perfect clone
        const originalCards = Array.from(track.children);
        originalCards.forEach(card => {
            const clone = card.cloneNode(true);
            track.appendChild(clone);
        });

        // Identify where the loop starts
        const firstClone = track.children[originalCards.length];
        const firstOriginal = track.children[0];
        
        let scrollSpeed = 1; 
        let isHovered = false;
        let isDragging = false;
        let startX, scrollLeft;

        // 2. Seamless Auto-Scroll
        function autoScroll() {
            if (!isHovered && !isDragging) {
                carousel.scrollLeft += scrollSpeed;
                
                // Calculates the EXACT pixel distance regardless of padding/gaps
                const resetPoint = firstClone.offsetLeft - firstOriginal.offsetLeft;
                
                if (carousel.scrollLeft >= resetPoint) {
                    carousel.scrollLeft = 0; 
                }
            }
            requestAnimationFrame(autoScroll);
        }
        autoScroll();

        // 3. Pause on Hover/Touch
        carousel.addEventListener('mouseenter', () => isHovered = true);
        carousel.addEventListener('mouseleave', () => {
            isHovered = false;
            isDragging = false;
        });
        carousel.addEventListener('touchstart', () => isHovered = true);
        carousel.addEventListener('touchend', () => {
            setTimeout(() => isHovered = false, 1500); 
            isDragging = false;
        });

        // 4. Drag-to-Scroll Logic
        const elementsToPreventDrag = carousel.querySelectorAll('a, video, img');
        elementsToPreventDrag.forEach(el => {
            el.addEventListener('dragstart', (e) => e.preventDefault());
        });

        carousel.addEventListener('mousedown', (e) => {
            isHovered = true;
            isDragging = false;
            carousel.style.cursor = 'grabbing';
            startX = e.pageX - carousel.offsetLeft;
            scrollLeft = carousel.scrollLeft;
        });

        carousel.addEventListener('mouseup', () => {
            isHovered = false;
            carousel.style.cursor = 'grab';
            
            // Loop backward seamlessly if aggressively dragged back
            const resetPoint = firstClone.offsetLeft - firstOriginal.offsetLeft;
            if (carousel.scrollLeft <= 0) {
                carousel.scrollLeft = resetPoint - 10;
            }
        });

        carousel.addEventListener('mousemove', (e) => {
            if (!isHovered || carousel.style.cursor !== 'grabbing') return;
            e.preventDefault();
            
            const x = e.pageX - carousel.offsetLeft;
            const walk = (x - startX); 
            
            if (Math.abs(walk) > 3) {
                isDragging = true; 
            }
            
            carousel.scrollLeft = scrollLeft - (walk * 2);
        });

        // Prevent links from firing during a drag
        const links = carousel.querySelectorAll('a.compact-reel-card');
        links.forEach(link => {
            link.addEventListener('click', (e) => {
                if (isDragging) {
                    e.preventDefault();
                }
            });
        });
    });
</script>

<script>
    document.addEventListener("DOMContentLoaded", function() {
        const videos = document.querySelectorAll('.reel-video');

        if ('IntersectionObserver' in window) {
            const videoObserver = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    const video = entry.target;
                    
                    if (entry.isIntersecting) {
                        video.playPromise = video.play();
                        if (video.playPromise !== undefined) {
                            video.playPromise.catch(error => {
                                console.log("Autoplay prevented or interrupted:", error);
                            });
                        }
                    } else {
                        if (video.playPromise !== undefined) {
                            video.playPromise.then(() => {
                                video.pause();
                            }).catch(error => {
                                console.log("Pause error:", error);
                            });
                        } else {
                            video.pause();
                        }
                    }
                });
            }, {
                threshold: 0.5 
            });

            videos.forEach(video => {
                videoObserver.observe(video);
            });
        } else {
            videos.forEach(video => video.setAttribute('autoplay', ''));
        }
    });
</script>
